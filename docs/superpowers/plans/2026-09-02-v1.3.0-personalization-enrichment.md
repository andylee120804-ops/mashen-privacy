// ── 确定性 PRNG（Park-Miller minimal standard，double 精度内精确无溢出）──

/** 31 位滚动哈希：h*31+c，h < 2^31，h*31 < 2^53，double 运算精确 */
function hashStr(s: string): number {
  let h: number = 0
  for (let i = 0; i < s.length; i++) {
    h = (h * 31 + s.charCodeAt(i)) % 2147483647
  }
  return h
}

/** Park-Miller PRNG：同 seed 序列恒定 */
class Rng {
  private s: number

  constructor(seed: number) {
    this.s = (seed % 2147483646) + 1
  }

  next(): number {
    this.s = (this.s * 16807) % 2147483647
    return this.s / 2147483647
  }
}

/** 从池中无放回抽 n 条 */
function pickN(pool: string[], n: number, rng: Rng): string[] {
  const arr: string[] = [...pool]
  const out: string[] = []
  const count = Math.min(n, arr.length)
  for (let i = 0; i < count; i++) {
    const j = i + Math.floor(rng.next() * (arr.length - i))
    const tmp = arr[i]
    arr[i] = arr[j]
    arr[j] = tmp
    out.push(arr[i])
  }
  return out
}

function pickNNum(pool: number[], n: number, rng: Rng): number[] {
  const arr: number[] = [...pool]
  const out: number[] = []
  const count = Math.min(n, arr.length)
  for (let i = 0; i < count; i++) {
    const j = i + Math.floor(rng.next() * (arr.length - i))
    const tmp = arr[i]
    arr[i] = arr[j]
    arr[j] = tmp
    out.push(arr[i])
  }
  return out
}

/** 加权抽签，返回命中下标 */
function pickWeighted(weights: number[], rng: Rng): number {
  let total = 0
  for (const w of weights) {
    total += w
  }
  let r = rng.next() * total
  for (let i = 0; i < weights.length; i++) {
    r -= weights[i]
    if (r < 0) {
      return i
    }
  }
  return weights.length - 1
}

export function generateShengxiaoFortune(dateStr: string, shengxiaoIdx: number): ShengxiaoFortune {
  const rng = new Rng(hashStr(dateStr + '|shengxiao|' + shengxiaoIdx))
  const stars = 1 + Math.floor(rng.next() * 5)
  const summary = SUMMARY_POOL[Math.floor(rng.next() * SUMMARY_POOL.length)]
  const paiyun = pickWeighted(PAIYUN_WEIGHTS, rng)
  const yi = pickN(YI_POOL, 3, rng)
  const ji = pickN(JI_POOL, 3, rng)
  // 牌运与宜忌呼应：旺/吉必有「宜打牌」，弱/忌必有「忌赌局」（用户核心诉求：明显标注适不适合打牌）
  if (paiyun <= 1 && !yi.includes('宜打牌')) {
    yi[0] = '宜打牌'
  }
  if (paiyun >= 3 && !ji.includes('忌赌局')) {
    ji[0] = '忌赌局'
  }
  const luckyNumbers = pickNNum(NUM_POOL, 2, rng)
  const luckyColor = LUCKY_COLORS[Math.floor(rng.next() * LUCKY_COLORS.length)]
  const result: ShengxiaoFortune = {
    stars: stars,
    summary: summary,
    paiyun: paiyun,
    yi: yi,
    ji: ji,
    luckyNumbers: luckyNumbers,
    luckyColor: luckyColor
  }
  return result
}

// ── 黄历（任意日期查询；与祈福页黄历卡同源同公式，同日结果一致）──

export function getAlmanacForDate(dateStr: string): AlmanacEntry {
  const d = new Date(dateStr + 'T00:00:00')
  const start = new Date(d.getFullYear(), 0, 0)
  const dayOfYear = Math.floor((d.getTime() - start.getTime()) / 86400000)
  return ALMANAC_TABLE[dayOfYear % ALMANAC_TABLE.length]
}

// ── 日期工具 ──

export function formatDate(d: Date): string {
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
}

export function dateOffsetStr(dateStr: string, days: number): string {
  const d = new Date(dateStr + 'T00:00:00')
  d.setDate(d.getDate() + days)
  return formatDate(d)
}

/** 返回 '日' '一' '二' ... '六' */
export function weekdayOf(dateStr: string): string {
  const d = new Date(dateStr + 'T00:00:00')
  const names: string[] = ['日', '一', '二', '三', '四', '五', '六']
  return names[d.getDay()]
}

/** 星级文本：stars=3 → '★★★☆☆' */
export function starText(stars: number): string {
  let s = ''
  for (let i = 0; i < stars; i++) {
    s += '★'
  }
  for (let i = stars; i < 5; i++) {
    s += '☆'
  }
  return s
}

  rng.next() // summary（池大小不同不影响后续序列一致性断言，仅跳过一次抽样）
  const paiyun = pickWeighted(PAIYUN_WEIGHTS, rng)
  const yi = pickN([...Array(YI_SIZE).keys()].map(String), 3, rng)
  const ji = pickN([...Array(JI_SIZE).keys()].map(String), 3, rng)
  if (paiyun <= 1 && !yi.includes('0')) yi[0] = '0' // 池[0] 即「宜打牌」
  if (paiyun >= 3 && !ji.includes('0')) ji[0] = '0' // 池[0] 即「忌赌局」
  const luckyNumbers = pickN(NUM_POOL, 2, rng)
  return { stars, paiyun, yi, ji, luckyNumbers }
}

let failed = 0
function assert(cond, msg) {
  if (!cond) {
    console.error('FAIL:', msg)
    failed++
  }
}

// 1. 确定性：同输入两次生成 deep-equal
for (let idx = 0; idx < 12; idx++) {
  const a = JSON.stringify(generateShengxiaoFortune('2026-09-02', idx))
  const b = JSON.stringify(generateShengxiaoFortune('2026-09-02', idx))
  assert(a === b, `determinism failed for idx=${idx}`)

// 2. 形状合法 + 牌运-宜忌呼应
for (let idx = 0; idx < 12; idx++) {

  assert(f.luckyNumbers.length === 2 && new Set(f.luckyNumbers).size === 2, 'luckyNumbers must be 2 unique')

// 3. 牌运档分布（366 天 × 12 属相 = 4392 样本，误差 ±5 个百分点）

const total = 366 * 12

// 界面定制：可配置入口的 Tab 开关与排序（v1.3.0 个性化显示）。

/** 可配置入口静态定义 */
export interface EntryDef {

  icon: string
  title: string

/** 入口运行时配置项 */
export interface EntryItem {
  id: string
  enabled: boolean   // 是否作为底部 Tab 显示
  order: number       // 排序权重（越小越靠前，Tab 与「我的入口」共用）

 * 入口池（id → 页面组件映射见 MainTabs/EntryShell）。
 * 默认 order：fortune 置顶「我的入口」第 1 条；默认 Tab = 求签 + 木鱼。

export const ENTRY_POOL: EntryDef[] = [
  { id: 'fortune', icon: '🗓️', title: '运势·黄历' },
  { id: 'qiuqian', icon: '🎋', title: '求签' },

  { id: 'leaderboard', icon: '🏆', title: '祈福排行榜' },

const PREF_NAME = 'mashen_entry_prefs'
const KEY_ENTRY_CONFIG = 'entryConfig'

const DEFAULT_ITEMS: EntryItem[] = [
  { id: 'fortune', enabled: false, order: 1 },
  { id: 'qiuqian', enabled: true, order: 2 },
  { id: 'muyu', enabled: true, order: 3 },
  { id: 'wishwall', enabled: false, order: 4 },
  { id: 'calendar', enabled: false, order: 5 },
  { id: 'leaderboard', enabled: false, order: 6 },
  { id: 'encyclopedia', enabled: false, order: 7 }
]

@Observed
export class EntryConfigVM {
  items: EntryItem[] = DEFAULT_ITEMS.map((it: EntryItem): EntryItem => {
    return { id: it.id, enabled: it.enabled, order: it.order }
  })
  private dataPrefs: preferences.Preferences | null = null

  async initialize(context: Context): Promise<void> {
    this.dataPrefs = await preferences.getPreferences(context, PREF_NAME)
    try {
      const savedStr = await this.dataPrefs.get(KEY_ENTRY_CONFIG, '') as string
      if (savedStr === '') {
        return
      }
      const saved: EntryItem[] = JSON.parse(savedStr)
      if (!Array.isArray(saved) || saved.length === 0) {
        return
      }
      // 以池为准合并：新增入口补默认值，已下线入口忽略
      this.items = ENTRY_POOL.map((def: EntryDef): EntryItem => {
        const found = saved.find((it: EntryItem): boolean => it.id === def.id)
        if (found) {
          return { id: found.id, enabled: found.enabled === true, order: found.order }
        }
        const fallback = DEFAULT_ITEMS.find((it: EntryItem): boolean => it.id === def.id)
        if (fallback) {
          return { id: fallback.id, enabled: fallback.enabled, order: fallback.order }
        }
        return { id: def.id, enabled: false, order: 99 }
      })
    } catch (err) {
      console.error('[EntryConfigVM] Failed to load prefs:', JSON.stringify(err))
    }
  }

  /** 开关入口的 Tab 显示；开启第 MAX_TABS+1 个时拒绝并返回 false */
  setEnabled(id: string, on: boolean): boolean {
    if (on && this.tabEnabledCount() >= MAX_TABS) {
      return false
    }
    this.items = this.items.map((it: EntryItem): EntryItem => {
      if (it.id === id) {
        return { id: it.id, enabled: on, order: it.order }
      }
      return it
    })
    this.persist()
    return true
  }

  /** 与全量排序列表中的相邻项交换 order；dir=-1 上移，1 下移 */
  swap(id: string, dir: number): void {
    const sorted = this.getSortedItems()
    const idx = sorted.findIndex((it: EntryItem): boolean => it.id === id)
    const j = idx + dir
    if (idx < 0 || j < 0 || j >= sorted.length) {
      return
    }
    const a = sorted[idx]
    const b = sorted[j]
    this.items = this.items.map((it: EntryItem): EntryItem => {
      if (it.id === a.id) {
        return { id: it.id, enabled: it.enabled, order: b.order }
      }
      if (it.id === b.id) {
        return { id: it.id, enabled: it.enabled, order: a.order }
      }
      return it
    })
    this.persist()
  }

  isFirst(id: string): boolean {
    const sorted = this.getSortedItems()
    return sorted.length > 0 && sorted[0].id === id
  }

  isLast(id: string): boolean {
    const sorted = this.getSortedItems()
    return sorted.length > 0 && sorted[sorted.length - 1].id === id
  }

  isTabOn(id: string): boolean {
    const it = this.items.find((item: EntryItem): boolean => item.id === id)
    return it ? it.enabled : false
  }

  tabEnabledCount(): number {
    return this.items.filter((it: EntryItem): boolean => it.enabled).length
  }

  /** 作为 Tab 显示的入口（按 order 升序），供 MainTabs 渲染 */
  getTabEntries(): EntryDef[] {
    const defs: EntryDef[] = []
    const enabled = [...this.items]
      .sort((a: EntryItem, b: EntryItem): number => a.order - b.order)
      .filter((it: EntryItem): boolean => it.enabled)
    for (const it of enabled) {
      const def = ENTRY_POOL.find((d: EntryDef): boolean => d.id === it.id)
      if (def) {
        defs.push(def)
      }
    }
    return defs
  }

  /** 全量配置项（按 order 升序），供「界面定制」与「我的入口」渲染 */
  getSortedItems(): EntryItem[] {
    return [...this.items].sort((a: EntryItem, b: EntryItem): number => a.order - b.order)
  }

  getIcon(id: string): string {
    const def = ENTRY_POOL.find((d: EntryDef): boolean => d.id === id)
    return def ? def.icon : ''
  }

  getTitle(id: string): string {
    const def = ENTRY_POOL.find((d: EntryDef): boolean => d.id === id)
    return def ? def.title : ''
  }

  private async persist(): Promise<void> {
    if (!this.dataPrefs) {
      return
    }
    try {
      await this.dataPrefs.put(KEY_ENTRY_CONFIG, JSON.stringify(this.items))
      await this.dataPrefs.flush()
    } catch (err) {
      console.error('[EntryConfigVM] Failed to persist:', JSON.stringify(err))
    }
  }
}

  /** 场景祝福行：`✨ {场景名} · {场景祝福语}`，按日期+总祈福次数确定性取 */
  private getSceneLine(): string {
    const scene = SCENE_DEFS.find((s: SceneDef): boolean => s.id === this.viewModel.sceneId)
    if (!scene || scene.blessings.length === 0) {
      return ''
    }
    const dayOfYear = this.getDayOfYear()
    const idx = (dayOfYear + this.viewModel.totalPrayers) % scene.blessings.length
    return `✨ ${scene.name} · ${scene.blessings[idx]}`
  }

  private getDayOfYear(): number {
    const now = new Date()
    const start = new Date(now.getFullYear(), 0, 0)
    return Math.floor((now.getTime() - start.getTime()) / 86400000)
  }

1. **ForEach 渲染 TabContent**：HarmonyOS 支持 ForEach 生成 TabContent（动态 Tab 的文档模式）。若真机上 TabContent 增删异常，改为 `getTabEntries()` 顺序驱动的排序 if 链，TabBarBuilder 索引逻辑不变。

3. **test-fortune-rng.cjs 与 FortuneTypes.ets 必须同步改**：PRNG/权重/呼应逻辑任一变更都要同步两处，否则验证脚本失去意义。
4. **Leaderboard/EncyclopediaPage 从 main_pages.json 移除后只能经 EntryShell 进入**：如需回退，恢复 @Entry 与路由表即可。
